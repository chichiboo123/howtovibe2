-- 1) 표 만들기
create table if not exists public.checkins (
  id         bigint generated always as identity primary key,
  room_code  text not null,                    -- 수업 코드 (예: 5-3)
  emotion    text not null,                    -- 고른 감정
  note       text,                             -- 한 줄 메모 (선택)
  created_at timestamptz not null default now()
);

create index if not exists checkins_room_time_idx
  on public.checkins (room_code, created_at desc);

-- 2) 문지기(RLS) 켜기
alter table public.checkins enable row level security;

-- 3) 읽기 허용 — 학급 전체 집계를 보여주기 위해
drop policy if exists "checkins_read" on public.checkins;
create policy "checkins_read" on public.checkins
  for select using (true);

-- 4) 쓰기 허용 — 길이까지 함께 확인합니다
drop policy if exists "checkins_insert" on public.checkins;
create policy "checkins_insert" on public.checkins
  for insert with check (
    length(room_code) between 1 and 40
    and length(emotion) between 1 and 20
    and (note is null or length(note) <= 100)
  );

-- 5) 실시간(Realtime) 켜기 — 대시보드에서 따로 켤 필요가 없습니다
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime'
      and schemaname = 'public'
      and tablename = 'checkins'
  ) then
    alter publication supabase_realtime add table public.checkins;
  end if;
end $$;

var SHEET_NAME = 'checkins';
var HEADERS = ['created_at', 'room_code', 'emotion', 'note'];
var TZ = 'Asia/Seoul';

/** 시트가 없으면 만들고 머리글까지 자동으로 넣습니다 */
function ensureSheet_() {
  var ss = SpreadsheetApp.getActive();
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
    sh.setFrozenRows(1);
    sh.setColumnWidth(1, 170);
  }
  return sh;
}

/** 처음 한 번 실행하세요 (잊어도 첫 저장 때 자동으로 만들어집니다) */

  Logger.log('준비 완료 — ' + SHEET_NAME + ' 시트와 머리글을 만들었습니다.');

/** 감정 하나를 저장하고, 새 결과를 돌려줍니다 */
function saveCheckin(payload) {
  var room = String((payload && payload.room) || '').trim().slice(0, 40);
  var emotion = String((payload && payload.emotion) || '').trim().slice(0, 20);
  var note = String((payload && payload.note) || '').trim().slice(0, 100);
  if (!room) throw new Error('수업 코드가 없습니다.');
  if (!emotion) throw new Error('감정을 고르지 않았습니다.');

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);            // 여러 명이 동시에 눌러도 줄이 겹치지 않게
  try {
    ensureSheet_().appendRow([new Date(), room, emotion, note]);
  } finally {
    lock.releaseLock();
  }
  return getRows(room);
}

/** 같은 수업 코드의 '오늘' 데이터만 돌려줍니다 */
function getRows(roomCode) {
  var room = String(roomCode || '').trim().slice(0, 40);
  var sh = ensureSheet_();
  var out = [];
  if (!room || sh.getLastRow() < 2) return out;

  var values = sh.getRange(2, 1, sh.getLastRow() - 1, HEADERS.length).getValues();
  var today = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');

  for (var i = values.length - 1; i >= 0; i--) {      // 최신 줄부터 봅니다
    var r = values[i];
    if (!r[0] || String(r[1]).trim() !== room) continue;
    var when = new Date(r[0]);
    if (Utilities.formatDate(when, TZ, 'yyyy-MM-dd') !== today) continue;
    out.push({ emotion: String(r[2]), note: String(r[3] || ''), at: when.toISOString() });
    if (out.length >= 1000) break;
  }
  return out;
}

function callServer(name, arg) {
  return new Promise(function (resolve, reject) {
    if (!(window.google && google.script && google.script.run)) {
      reject(new Error('Apps Script 웹앱 주소(/exec)에서 열어주세요.'));
      return;
    }
    var runner = google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(function (e) {
        reject(new Error(e && e.message ? e.message : '서버 오류'));
      });
    if (arg === undefined) runner[name]();
    else runner[name](arg);
  });
}

function summarize(rows) {
  var counts = {}, notes = [], total = 0;
  EMOTIONS.forEach(function (e) { counts[e.name] = 0; });
  (rows || []).forEach(function (r) {
    if (counts[r.emotion] === undefined) return;   /* 목록에 없는 값은 셈에서 제외 */
    counts[r.emotion] += 1;
    total += 1;
    if (r.note) notes.push({ text: String(r.note), at: r.at });
  });
  notes.sort(function (a, b) { return new Date(b.at) - new Date(a.at); });
  return { counts: counts, notes: notes.slice(0, 30), total: total, updatedAt: nowText() };
}

function startUpdates() {
  if (stopWatch) { stopWatch(); stopWatch = null; }
  if (timer) { clearInterval(timer); timer = null; }

  if (Backend.realtime) {
    setLive('ok', '실시간 연결');
    stopWatch = Backend.watch(room, function (rows) {
      if (rows) renderSummary(summarize(rows));
      else refresh();
    });
    refresh();
  } else {
    refresh();
    timer = setInterval(function () {
      if (document.hidden) return;         /* 탭이 안 보일 때는 쉬어갑니다 */
      refresh();
    }, Backend.pollMs);
    document.addEventListener('visibilitychange', function () {
      if (!document.hidden && room) refresh();
    });
  }
}

var db = null;
function store() {
  if (!db) db = getFirestore(initializeApp(firebaseConfig));
  return db;
}
/* 수업 코드와 날짜를 한 칸에 합쳐 둡니다.
   그래야 조건이 하나뿐이어서 색인(index)을 따로 만들 일이 없습니다. */
function roomDay(room) { return room + '__' + dayKey(); }

function toRows(snap) {
  var rows = [];
  snap.forEach(function (doc) {
    var v = doc.data();
    rows.push({
      emotion: v.emotion,
      note: v.note,
      at: (v.created_at && v.created_at.toDate) ? v.created_at.toDate() : new Date()
    });
  });
  return rows;
}

var Backend = {
  realtime: true,
  pollMs: 0,

  configProblem: function () {
    if (!firebaseConfig.projectId || firebaseConfig.projectId.indexOf('여기에') === 0) {
      return '연결 설정이 아직 비어 있습니다. 코드 맨 위의 firebaseConfig 를 Firebase 콘솔 값으로 바꿔주세요.';
    }
    return '';
  },

  save: function (rec) {
    return addDoc(collection(store(), 'checkins'), {
      room_day: roomDay(rec.room),
      room_code: rec.room,
      emotion: rec.emotion,
      note: rec.note || '',
      created_at: Timestamp.now()
    }).then(function () { return null; });   /* 실시간 구독이 곧 화면을 갱신합니다 */
  },

  load: function (room) {
    var q = query(collection(store(), 'checkins'),
                  where('room_day', '==', roomDay(room)));
    return getDocs(q).then(toRows);
  },

  watch: function (room, onRows) {
    var q = query(collection(store(), 'checkins'),
                  where('room_day', '==', roomDay(room)));
    /* onSnapshot 이 돌려주는 함수를 부르면 구독이 끊깁니다.
       끊지 않으면 읽기 횟수가 계속 늘어나 무료 한도를 빨리 씁니다. */
    return onSnapshot(q,
      function (snap) { onRows(toRows(snap)); },
      function () { setLive('bad', '연결 끊김 · 새로고침'); });
  },

  baseUrl: function () {
    return Promise.resolve(location.origin + location.pathname);
  },
  roomFromUrl: function () {
    return Promise.resolve(new URLSearchParams(location.search).get('room') || '');

const CORS = {

function reply(data, status) {

/** 같은 수업 코드의 '오늘'(한국 시간) 데이터만 돌려줍니다 */
async function readRows(env, room) {
  const out = await env.DB.prepare(
    "select emotion, note, created_at from checkins " +
    " where room_code = ? " +
    "   and date(created_at, '+9 hours') = date('now', '+9 hours') " +
    " order by id desc limit 1000"
  ).bind(room).all();

  return (out.results || []).map(function (r) {
    return {
      emotion: r.emotion,
      note: r.note || '',
      at: String(r.created_at).replace(' ', 'T') + 'Z'   // UTC임을 알려줍니다
    };
  });
}

export default {
  async fetch(request, env) {
    if (request.method === 'OPTIONS') return new Response(null, { headers: CORS });

    const url = new URL(request.url);
    if (!url.pathname.startsWith('/api/checkins')) {
      return reply({ ok: false, error: '없는 주소입니다.' }, 404);
    }
    if (!env.DB) {
      return reply({ ok: false, error: 'D1 연결(Binding)이 없습니다. 변수 이름을 DB 로 만들고 다시 배포하세요.' }, 500);
    }

    try {
      if (request.method === 'POST') {
        const body = await request.json();
        const room = String(body.room_code || '').trim().slice(0, 40);
        const emotion = String(body.emotion || '').trim().slice(0, 20);
        const note = String(body.note || '').trim().slice(0, 100);
        if (!room || !emotion) {
          return reply({ ok: false, error: '수업 코드와 감정이 필요합니다.' }, 400);
        }
        await env.DB.prepare(
          'insert into checkins (room_code, emotion, note) values (?, ?, ?)'
        ).bind(room, emotion, note).run();

        return reply({ ok: true, rows: await readRows(env, room) });
      }

      if (request.method === 'GET') {
        const room = String(url.searchParams.get('room_code') || '').trim().slice(0, 40);
        if (!room) return reply({ ok: false, error: '수업 코드가 필요합니다.' }, 400);
        return reply({ ok: true, rows: await readRows(env, room) });
      }
    } catch (err) {
      return reply({ ok: false, error: String((err && err.message) || err) }, 500);
    }

    return reply({ ok: false, error: '허용하지 않는 방식입니다.' }, 405);
  }
};

function api(path, options) {
  var base = String(WORKER_URL).trim().replace(/\/+$/, '');
  /* 헤더를 일부러 붙이지 않습니다. 붙이면 CORS 사전 요청 때문에 막힐 수 있습니다. */

    });
  });

function summarize(rows) {
  var counts = {}, notes = [], total = 0;
  EMOTIONS.forEach(function (e) { counts[e.name] = 0; });
  (rows || []).forEach(function (r) {
    if (counts[r.emotion] === undefined) return;   /* 목록에 없는 값은 셈에서 제외 */

            학생에게는 로그인 화면이 전혀 나타나지 않지만, 앱은 내부적으로 사용자를 구분합니다.
            콘솔의 Authentication → 로그인 방법 → 익명을 켜야 작동합니다.

          

            

rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {

    match /checkins/{id} {
      allow read: if true;

      // 저장할 때 본인 uid를 반드시 함께 넣게 강제합니다
      allow create: if request.auth != null
                    && request.resource.data.uid == request.auth.uid;

      // 본인이 쓴 문서만 수정·삭제할 수 있습니다
      allow update, delete: if request.auth != null
                            && resource.data.uid == request.auth.uid;
    }
  }
}

// /api 로 시작하면 데이터, 나머지 주소는 화면(HTML)을 돌려줍니다
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (url.pathname.startsWith('/api')) return handleApi(request, env);
    return new Response(APP_HTML, {
      headers: { 'content-type': 'text/html; charset=utf-8' }
    });
  }
};
