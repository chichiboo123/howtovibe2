상단 메뉴 확장 프로그램 → Apps Script를 열고, 기본 코드를 전부 지운 뒤 아래를 붙여넣고 저장합니다.

// 우리 반 감정 체크인 — Apps Script 웹앱
// 시트 이름 : checkins   /   1행 머리글 : created_at | room_code | emotion | note
const SHEET = 'checkins';

// 저장 : 앱에서 감정을 누르면 시트에 한 줄이 쌓입니다
function doPost(e) {
  const d = JSON.parse(e.postData.contents);
  if (!d.room_code || !d.emotion) return json({ ok: false });
  SpreadsheetApp.getActive().getSheetByName(SHEET).appendRow([
    new Date(),
    String(d.room_code).slice(0, 40),
    String(d.emotion).slice(0, 20),
    String(d.note || '').slice(0, 200)
  ]);
  return json({ ok: true });
}

// 불러오기 : 같은 수업 코드의 '오늘' 데이터만 감정별로 세어서 돌려줍니다
function doGet(e) {
  const room = e.parameter.room_code || '';
  const rows = SpreadsheetApp.getActive().getSheetByName(SHEET)
                 .getDataRange().getValues().slice(1);
  const today = Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd');
  const counts = {}, notes = [];
  rows.forEach(function (r) {
    if (!r[0] || r[1] !== room) return;
    if (Utilities.formatDate(new Date(r[0]), 'Asia/Seoul', 'yyyy-MM-dd') !== today) return;
    counts[r[2]] = (counts[r[2]] || 0) + 1;
    if (r[3]) notes.push(r[3]);        // 작성자 정보 없이 내용만
  });
  return json({ counts: counts, notes: notes });
}

function json(o) {
  return ContentService.createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}

// 수정·삭제 기능은 일부러 만들지 않았습니다.
// 없는 기능은 뚫릴 수도 없습니다. 삭제가 필요하면 교사가 시트에서 행을 지웁니다.

이 코드가 하는 일은 두 가지입니다. 한 줄 쌓기(doPost)와, 오늘 것만 세어서 돌려주기(doGet). 집계를 서버에서 해 주기 때문에 앱은 받은 숫자를 그리기만 하면 됩니다.

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
  'Access-Control-Allow-Headers': 'content-type'
};

export default {
  async fetch(request, env) {
    if (request.method === 'OPTIONS') return new Response(null, { headers: CORS });
    const url = new URL(request.url);

    // 저장
    if (request.method === 'POST') {
      const d = await request.json();
      if (!d.room_code || !d.emotion) {
        return Response.json({ ok: false }, { status: 400, headers: CORS });
      }
      await env.DB.prepare(
        'insert into checkins (room_code, emotion, note) values (?, ?, ?)'
      ).bind(
        String(d.room_code).slice(0, 40),
        String(d.emotion).slice(0, 20),
        String(d.note || '').slice(0, 200)
      ).run();
      return Response.json({ ok: true }, { headers: CORS });
    }

    // 불러오기 — 같은 수업 코드의 '오늘' 데이터만 감정별로 집계
    const room = url.searchParams.get('room_code') || '';
    const { results } = await env.DB.prepare(
      "select emotion, count(*) as n from checkins " +
      "where room_code = ? and date(created_at, '+9 hours') = date('now', '+9 hours') " +
      "group by emotion"
    ).bind(room).all();
    return Response.json({ counts: results }, { headers: CORS });
  }
};
