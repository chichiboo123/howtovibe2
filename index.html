function saveCheckin(payload) {
  var room = String((payload && payload.room) || '').trim().slice(0, 40);
  var emotion = String((payload && payload.emotion) || '').trim().slice(0, 20);
  var note = String((payload && payload.note) || '').trim().slice(0, 100);
  if (!room) throw new Error('수업 코드가 없습니다.');
  if (!emotion) throw new Error('기분을 고르지 않았습니다.');

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);            // 여러 명이 동시에 눌러도 줄이 겹치지 않게
  try {
    ensureSheet_().appendRow([new Date(), sheetText_(room), sheetText_(emotion), sheetText_(note)]);
  } finally {
    lock.releaseLock();
  }
  return getRows({ room: room });
}

/**
 * 같은 수업 코드의 '하루치' 데이터를 돌려줍니다.
 * 날짜를 넘기지 않으면 오늘, 넘기면 그날의 기록을 돌려줍니다.
 *   예)  getRows({ room: '5-3', day: '2026-03-04' })
 */
function getRows(arg) {
  var room = '', day = '';
  if (arg && typeof arg === 'object') {
    room = String(arg.room || '').trim().slice(0, 40);
    day = String(arg.day || '').trim();
  } else {
    room = String(arg || '').trim().slice(0, 40);
  }
  var sh = ensureSheet_();
  var out = [];
  if (!room || sh.getLastRow() < 2) return out;

  var values = sh.getRange(2, 1, sh.getLastRow() - 1, HEADERS.length).getValues();
  var want = /^\d{4}-\d{2}-\d{2}$/.test(day)
    ? day
    : Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');

  for (var i = values.length - 1; i >= 0; i--) {      // 최신 줄부터 봅니다
    var r = values[i];
    if (!r[0] || String(r[1]).trim() !== room) continue;
    var when = new Date(r[0]);
    if (Utilities.formatDate(when, TZ, 'yyyy-MM-dd') !== want) continue;
    out.push({ emotion: String(r[2]), note: String(r[3] || ''), at: when.toISOString() });
    if (out.length >= 1000) break;
  }
  return out;
}

function sheetText_(value) {

/** 시트가 없으면 만들고 머리글까지 자동으로 넣습니다 */

/** 의견 카드 한 장을 붙이고, 새 목록을 돌려줍니다 */
function addCard(payload) {
  var room = String((payload && payload.room) || '').trim().slice(0, 40);
  var text = String((payload && payload.text) || '').trim().slice(0, 150);
  var color = String((payload && payload.color) || '');
  if (!room) throw new Error('수업 코드가 없습니다.');
  if (!text) throw new Error('의견을 적어주세요.');
  if (CARD_COLORS.indexOf(color) < 0) color = CARD_COLORS[0];

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);            // 여러 명이 동시에 눌러도 줄이 겹치지 않게
  try {
    ensureSheet_().appendRow([Utilities.getUuid(), new Date(), sheetText_(room), sheetText_(text), color, 0]);
  } finally {
    lock.releaseLock();
  }
  return getCards({ room: room });
}

/**
 * 공감(♥) 하나를 더합니다.
 * '읽고 → 1 더하고 → 다시 쓰기' 라서, 두 명이 동시에 누르면 하나가 사라질 수 있습니다.
 * 그래서 잠금(LockService)을 걸고 한 명씩 차례로 처리합니다.
 */
function likeCard(payload) {
  var id = String((payload && payload.id) || '').trim();
  var room = String((payload && payload.room) || '').trim().slice(0, 40);
  if (!id) throw new Error('카드를 찾지 못했습니다.');

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = ensureSheet_();
    var cell = sh.getRange('A:A').createTextFinder(id).matchEntireCell(true).findNext();
    // 같은 수업 코드의 카드일 때만 공감을 더합니다
    if (cell && String(sh.getRange(cell.getRow(), 3).getValue()).trim() === room) {
      var likeCell = sh.getRange(cell.getRow(), 6);
      likeCell.setValue((Number(likeCell.getValue()) || 0) + 1);
    }
  } finally {
    lock.releaseLock();
  }
  return getCards({ room: room });
}

/** 같은 수업 코드의 카드를 최신순으로 돌려줍니다 */
function getCards(arg) {
  var room = String((arg && arg.room) || '').trim().slice(0, 40);
  var sh = ensureSheet_();
  var out = [];
  if (!room || sh.getLastRow() < 2) return out;

  var values = sh.getRange(2, 1, sh.getLastRow() - 1, HEADERS.length).getValues();
  for (var i = values.length - 1; i >= 0; i--) {       // 최신 줄부터 봅니다
    var r = values[i];
    if (!r[0] || String(r[2]).trim() !== room) continue;
    out.push({
      id: String(r[0]),
      at: new Date(r[1]).toISOString(),
      text: String(r[3]),
      color: String(r[4]),
      likes: Number(r[5]) || 0
    });
    if (out.length >= MAX_CARDS) break;
  }
  return out;
}

function callServer(name, arg) {
  return new Promise(function (resolve, reject) {
    if (!(window.google && google.script && google.script.run)) {
      reject(new Error('Apps Script 웹앱 주소(/exec)에서 열어주세요.'));
      return;
    }
    var runner = google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(function (e) {
        reject(new Error(e && e.message ? e.message : '서버 오류'));
      });
    if (arg === undefined) runner[name]();
    else runner[name](arg);
  });
}

/**
 * 화면이 부르는 함수.
 * 받아 둔 날씨가 있으면 그것을, 없으면 새로 받아서 돌려줍니다.
 * 반 전체가 동시에 열어도 Open-Meteo 에는 보통 20분에 한 번만 물어봅니다.
 */
function getWeather() {
  var cache = CacheService.getScriptCache();
  var saved = cache.get(CACHE_KEY);
  if (saved) return JSON.parse(saved);

  // 캐시가 비었을 때 여러 명이 동시에 열어도 한 명만 바깥에 물어보게 잠급니다
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    saved = cache.get(CACHE_KEY);              // 기다리는 동안 다른 사람이 받아 왔을 수 있습니다
    if (saved) return JSON.parse(saved);
    var data = fetchWeather_();
    cache.put(CACHE_KEY, JSON.stringify(data), CACHE_MIN * 60);
    return data;
  } finally {
    lock.releaseLock();
  }
}

/**
 * Open-Meteo 에 15개 도시를 '한 번에' 물어봅니다.
 * 도시마다 따로 물어보면 연수처럼 여러 명이 동시에 setup 을 누를 때
 * 요청이 몰려 거절될 수 있습니다. 위도·경도를 쉼표로 이어 붙이면 한 번이면 됩니다.
 */
function fetchWeather_() {
  var lats = CITIES.map(function (c) { return c.lat; }).join(',');
  var lons = CITIES.map(function (c) { return c.lon; }).join(',');
  var zones = CITIES.map(function () { return 'auto'; }).join(',');   // 도시마다 현지 시간대
  var url = 'https://api.open-meteo.com/v1/forecast' +
    '?latitude=' + lats + '&longitude=' + lons +
    '&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m,is_day' +
    '&daily=temperature_2m_max,temperature_2m_min' +
    '&timezone=' + zones + '&forecast_days=1';

  var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) {
    throw new Error('날씨 서버에 연결하지 못했습니다. 잠시 뒤 다시 시도해주세요.');
  }
  var list = JSON.parse(res.getContentText());
  if (!Array.isArray(list)) list = [list];       // 도시가 하나뿐이면 목록이 아니라 하나로 옵니다

  var cities = CITIES.map(function (c, i) {      // 답은 CITIES 와 같은 순서로 옵니다
    var j = list[i];
    if (!j || !j.current || !j.daily) return { name: c.name, country: c.country, error: true };
    return {
      name: c.name,
      country: c.country,
      temp: Math.round(j.current.temperature_2m),
      humidity: j.current.relative_humidity_2m,
      wind: Math.round(j.current.wind_speed_10m / 3.6 * 10) / 10,   // km/h → m/s
      code: j.current.weather_code,
      isDay: j.current.is_day === 1,
      max: Math.round(j.daily.temperature_2m_max[0]),
      min: Math.round(j.daily.temperature_2m_min[0]),
      offset: j.utc_offset_seconds                                  // 현지 시각 계산용
    };
  });
  return { updatedAt: new Date().toISOString(), cities: cities };
}

function callServer(name, arg) {
  return new Promise(function (resolve, reject) {
    if (!(window.google && google.script && google.script.run)) {
      reject(new Error('Apps Script 웹앱 주소(/exec)에서 열어주세요.'));
      return;
    }
    var runner = google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(function (e) {
        reject(new Error(e && e.message ? e.message : '서버 오류'));
      });
    if (arg === undefined) runner[name]();
    else runner[name](arg);
  });
}

function localTime(offsetSec) {
  var d = new Date(Date.now() + offsetSec * 1000);
  var h = d.getUTCHours(), m = d.getUTCMinutes();
  var ampm = h < 12 ? '오전' : '오후';
  var h12 = h % 12 === 0 ? 12 : h % 12;
  return ampm + ' ' + h12 + ':' + pad(m);
}

/* 받침이 있으면 '과', 없으면 '와' */
function gwa(word) {
  var c = word.charCodeAt(word.length - 1);
  if (c < 0xAC00 || c > 0xD7A3) return '와';
  return (c - 0xAC00) % 28 ? '과' : '와';
}

/** 글 한 편을 저장하고, 오늘 올라온 글 목록을 돌려줍니다 */
function saveWriting(payload) {
  var room = String((payload && payload.room) || '').trim().slice(0, 40);
  var topic = String((payload && payload.topic) || '').trim().slice(0, 80);
  var pen = String((payload && payload.pen) || '').trim().slice(0, 20);
  var title = String((payload && payload.title) || '').trim().slice(0, 40);
  var body = String((payload && payload.body) || '').trim().slice(0, 2000);
  if (!room) throw new Error('수업 코드가 없습니다.');
  if (!pen) throw new Error('필명을 적어주세요.');
  if (!body) throw new Error('글을 적어주세요.');

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    ensureSheet_().appendRow([new Date(), sheetText_(room), sheetText_(topic), sheetText_(pen), sheetText_(title), sheetText_(body)]);
  } finally {
    lock.releaseLock();
  }
  return getToday({ room: room });
}

/** 같은 수업 코드에 오늘 올라온 글을 최신순으로 돌려줍니다 */
function getToday(arg) {
  var room = String((arg && arg.room) || '').trim().slice(0, 40);
  var today = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd');
  return readRows_(room).filter(function (w) {
    return Utilities.formatDate(w.when, TZ, 'yyyy-MM-dd') === today;
  }).reverse().map(function (w) {
    return { at: w.when.toISOString(), topic: w.topic, pen: w.pen, title: w.title, body: w.body };
  });
}

/** 시트의 글을 오래된 순서로 읽습니다 (room 이 비어 있으면 전부) */
function readRows_(room) {
  var sh = ensureSheet_();
  if (sh.getLastRow() < 2) return [];
  var values = sh.getRange(2, 1, sh.getLastRow() - 1, HEADERS.length).getValues();
  var out = [];
  values.forEach(function (r) {
    if (!r[0] || !r[5]) return;
    if (room && String(r[1]).trim() !== room) return;
    out.push({ when: new Date(r[0]), room: String(r[1]), topic: String(r[2] || ''),
               pen: String(r[3]), title: String(r[4] || ''), body: String(r[5]) });
  });
  out.sort(function (a, b) { return a.when - b.when; });
  return out;
}

    ui = SpreadsheetApp.getUi();

  if (ask.getSelectedButton() !== ui.Button.OK) return;

    ui.alert('묶을 글이 없습니다. 수업 코드를 확인해주세요.');

function callServer(name, arg) {
  return new Promise(function (resolve, reject) {
    if (!(window.google && google.script && google.script.run)) {
      reject(new Error('Apps Script 웹앱 주소(/exec)에서 열어주세요.'));
      return;
    }
    var runner = google.script.run
      .withSuccessHandler(resolve)
      .withFailureHandler(function (e) {
        reject(new Error(e && e.message ? e.message : '서버 오류'));
      });
    if (arg === undefined) runner[name]();
    else runner[name](arg);
  });
}
